### Validation Schema Updates

**File**: `src/lib/validation/important-dates-schema.ts`

Update PE3 validation to require time:

```typescript
export const createImportantDateSchema = z.object({
  category: z.string().min(1),
  date_value: z.string().min(1),
  time_value: z.string().nullable().refine(
    (value, ctx) => {
      // If category is PE3, time must be provided
      if (ctx.parent.category === 'PE3 Dates' && !value) {
        return false;
      }
      return true;
    },
    { message: "Time is required for PE3 dates" }
  ),
  date_description: z.string().optional(),
  // ... other fields
});
```

### Component Test Patterns

**Pattern 1: Field Order Test**
```typescript
test('important dates form renders fields in correct order', () => {
  render(<AddImportantDateModal />);
  
  const fields = screen.getAllByRole('textbox').concat(
    screen.getAllByRole('combobox')
  );
  
  expect(fields[0]).toHaveAttribute('name', 'category');
  expect(fields[1]).toHaveAttribute('name', 'date_value');
  expect(fields[2]).toHaveAttribute('name', 'time_value');
  expect(fields[3]).toHaveAttribute('name', 'date_description');
});
```

**Pattern 2: Auto-Population Test**
```typescript
test('description auto-populates from date and time', async () => {
  const { user } = render(<AddImportantDateModal />);
  
  await user.selectOptions(screen.getByRole('combobox', { name: /category/i }), 'PE3 Dates');
  await user.type(screen.getByLabelText(/date/i), '2025-03-15');
  await user.type(screen.getByLabelText(/time/i), '14:30');
  
  expect(screen.getByLabelText(/description/i)).toHaveValue('15 mars 2025 14:30');
});
```

**Pattern 3: Validation Error Test**
```typescript
test('PE3 without time shows validation error', async () => {
  const { user } = render(<AddImportantDateModal />);
  
  await user.selectOptions(screen.getByRole('combobox'), 'PE3 Dates');
  await user.type(screen.getByLabelText(/date/i), '2025-03-15');
  // Intentionally skip time field
  
  await user.click(screen.getByRole('button', { name: /save/i }));
  
  expect(await screen.findByText(/time is required for PE3 dates/i)).toBeInTheDocument();
});
```
