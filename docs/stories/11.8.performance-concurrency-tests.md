### Key Test Patterns

**Pattern 1: API Response Time Benchmark**
```typescript
import { bench } from 'vitest';

bench('POST /api/employees response time', async () => {
  const employee = createTestEmployee();
  const start = performance.now();
  
  await fetch('/api/employees', {
    method: 'POST',
    body: JSON.stringify(employee)
  });
  
  const duration = performance.now() - start;
  expect(duration).toBeLessThan(100); // p95 target
}, { iterations: 100 });
```

**Pattern 2: Table Rendering Benchmark**
```typescript

bench('render 1000 employees', () => {

```

**Pattern 3: Concurrency Load Test (k6)**
```javascript
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  scenarios: {
    capacity_assignment: {
      executor: 'constant-vus',
      vus: 100,
      duration: '10s',
    },
  },
};

export default function () {
  const employee = {
    first_name: `Test${__VU}`,
    surname: 'Employee',
    omc_date: '2025-03-08' // Last spot
  };
  
  const res = http.post('http://localhost:3000/api/employees', JSON.stringify(employee));
  
  // Expect 409 for all but one request (capacity full)
  check(res, {
    'status is 201 or 409': (r) => r.status === 201 || r.status === 409,
  });
  
  sleep(0.1);
}
```

**Pattern 4: Memory Leak Test**
```typescript
test('no memory leak over 1 hour', async () => {
  const initialMemory = process.memoryUsage().heapUsed;
  const startTime = Date.now();
  
  // Run operations for 1 hour
  while (Date.now() - startTime < 60 * 60 * 1000) {
    // Create and delete employees repeatedly
    const employee = await createEmployee(createTestEmployee());
    await deleteEmployee(employee.id);
    
    // Check memory every 5 minutes
    if ((Date.now() - startTime) % (5 * 60 * 1000) === 0) {
      const currentMemory = process.memoryUsage().heapUsed;
      const memoryGrowth = currentMemory - initialMemory;
      
      // Memory shouldn't grow more than 50MB
      expect(memoryGrowth).toBeLessThan(50 * 1024 * 1024);
    }
  }
}, { timeout: 60 * 60 * 1000 + 10000 }); // 1 hour + buffer
```

**Pattern 5: Real-time Sync Latency Test**
```typescript
test('real-time sync latency <2s', async () => {
  const updates = [];
  const subscription = setupRealTimeSubscription('employees', (payload) => {
    updates.push({ ...payload, receivedAt: Date.now() });
  });
  
  const sentAt = Date.now();
  await createEmployee(createTestEmployee());
  
  // Wait for update
  await waitFor(() => updates.length > 0, { timeout: 3000 });
  
  const latency = updates[0].receivedAt - sentAt;
  expect(latency).toBeLessThan(2000); // <2s requirement
  
  subscription.unsubscribe();
});
```

### Test Utilities to Create

```typescript
// tests/performance/helpers/performance-helpers.ts

export function generateEmployees(count: number) {
  return Array.from({ length: count }, (_, i) => ({
    id: `emp-${i}`,
    first_name: `Employee${i}`,
    surname: 'Test',
    ssn: `19900101${String(i).padStart(4, '0')}`,
    rank: i % 2 === 0 ? 'SEV' : 'CHEF',
    gender: i % 2 === 0 ? 'Man' : 'Kvinna',
    hire_date: '2025-01-01',
    omc_date: i < 20 ? '2025-03-08' : null,
  }));
}

export function measureRenderTime(component: React.ReactElement) {
  const start = performance.now();
  render(component);
  return performance.now() - start;
}

export async function measureAPILatency(
  method: string,
  endpoint: string,
  body?: any
) {
  const start = performance.now();
  await fetch(endpoint, { method, body: body ? JSON.stringify(body) : undefined });
  return performance.now() - start;
}

export function calculatePercentile(values: number[], percentile: number) {
  const sorted = values.sort((a, b) => a - b);
  const index = Math.ceil((percentile / 100) * sorted.length) - 1;
  return sorted[index];
}

export async function runLoadTest(
  operation: () => Promise<void>,
  concurrentUsers: number,
  duration: number
) {
  const startTime = Date.now();
  const results = { success: 0, error: 0, latencies: [] };
  
  const users = Array.from({ length: concurrentUsers }, async () => {
    while (Date.now() - startTime < duration) {
      try {
        const start = performance.now();
        await operation();
        results.latencies.push(performance.now() - start);
        results.success++;
      } catch (err) {
        results.error++;
      }
    }
  });
  
  await Promise.all(users);
  
  return {
    ...results,
    p50: calculatePercentile(results.latencies, 50),
    p95: calculatePercentile(results.latencies, 95),
    p99: calculatePercentile(results.latencies, 99),
    errorRate: results.error / (results.success + results.error),
  };
}
```
