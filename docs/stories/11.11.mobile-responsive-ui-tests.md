### Responsive Test Pattern

**Pattern 1: Viewport Simulation Test**
```typescript
import { render } from '@testing-library/react';
import { setViewportSize } from '@/tests/helpers/responsive-test-helpers';

test('buttons stack vertically on mobile', () => {
  setViewportSize(375, 667); // iPhone SE
  
  const { container } = render(<DashboardButtons />);
  const buttonContainer = container.querySelector('.button-group');
  
  expect(buttonContainer).toHaveClass('flex-col');
  expect(buttonContainer).not.toHaveClass('flex-row');
});
```

**Pattern 2: Touch Target Validation Test**
```typescript
test('all buttons meet 44px touch target minimum', () => {
  setViewportSize(375, 667);
  
  const { getAllByRole } = render(<DashboardButtons />);
  const buttons = getAllByRole('button');
  
  buttons.forEach(button => {
    const rect = button.getBoundingClientRect();
    expect(rect.height).toBeGreaterThanOrEqual(44);
    expect(rect.width).toBeGreaterThanOrEqual(44);
  });
});
```

**Pattern 3: Conditional Rendering Test**
```typescript
test('envelope icon hidden on mobile', () => {
  setViewportSize(375, 667); // Mobile
  const { queryByTestId } = render(<UserCard />);
  expect(queryByTestId('envelope-icon')).not.toBeInTheDocument();
});

test('envelope icon visible on desktop', () => {
  setViewportSize(1024, 768); // Desktop
  const { getByTestId } = render(<UserCard />);
  expect(getByTestId('envelope-icon')).toBeInTheDocument();
});
```
