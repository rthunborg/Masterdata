## Context

The database has multiple constraints that enforce business rules at the database level:

**Database Constraints:**
- `remaining_spots` never negative (CHECK constraint)
- `remaining_spots` <= `max_spots` (CHECK constraint)
- Gender enum constraint (CHECK constraint)
- Rank enum constraint (CHECK constraint)
- PE3 date uniqueness (UNIQUE constraint)
- Foreign key integrity (employees → important_dates)
- Transaction atomicity (spots + assigned_employees)

**Current State:**
- ✅ Constraints implemented in database schema
- ❌ No tests validate constraint enforcement
- ❌ No tests for transaction rollback scenarios
- ❌ No tests for foreign key cascades

### Key Test Patterns

**Pattern 1: Constraint Violation Test**
```typescript
test('should reject negative remaining_spots', async () => {
  const date = await createTestDate({ remaining_spots: 0 });
  
  // Attempt to set negative value
  await expect(
    supabase
      .from('important_dates')
      .update({ remaining_spots: -1 })
      .eq('id', date.id)
  ).rejects.toThrow(/remaining_spots_non_negative/);
});
```

**Pattern 2: Enum Violation Test**
```typescript
test('should reject invalid gender enum', async () => {
  await expect(
    supabase
      .from('employees')
      .insert({ ...createTestEmployee(), gender: 'male' })
  ).rejects.toThrow(/gender_enum/);
});
```

**Pattern 3: Uniqueness Violation Test**
```typescript
test('should reject duplicate PE3 date', async () => {
  const pe3Date = '2025-04-20';
  
  // Create first employee with PE3 date
  await createTestEmployee({ pe3_date: pe3Date });
  
  // Attempt to create second employee with same PE3 date
  await expect(
    createTestEmployee({ pe3_date: pe3Date })
  ).rejects.toThrow(/unique_pe3_date/);
  
  // Verify error code
  try {
    await createTestEmployee({ pe3_date: pe3Date });
  } catch (err) {
    expect(err.code).toBe('23505'); // Unique violation
  }
});
```

**Pattern 4: Transaction Atomicity Test**
```typescript
test('should rollback both employee and capacity on failure', async () => {
  const date = await createTestDate({ remaining_spots: 10 });
  
  // Mock database error after employee creation but before capacity decrement
  vi.spyOn(supabase.rpc, 'update_date_spots').mockRejectedValueOnce(
    new Error('DB Error')
  );
  
  await expect(
    createEmployee({ ...createTestEmployee(), omc_date: date.date_value })
  ).rejects.toThrow();
  
  // Verify rollback: no employee created, capacity unchanged
  const employees = await getEmployees();
  expect(employees).toHaveLength(0);
  
  const updatedDate = await getDateById(date.id);
  expect(updatedDate.remaining_spots).toBe(10); // Unchanged
});
```

**Pattern 5: Concurrent Write Test**
```typescript
test('should handle concurrent capacity updates correctly', async () => {
  const date = await createTestDate({ remaining_spots: 10 });
  
  // Two concurrent decrements
  const [result1, result2] = await Promise.all([
    decrementCapacity(date.id),
    decrementCapacity(date.id)
  ]);
  
  // Both should succeed
  expect(result1).toBe(9);
  expect(result2).toBe(8);
  
  // Final value should be 8 (two decrements from 10)

  expect(finalDate.remaining_spots).toBe(8);
});
```

**Pattern 6: Cascading Delete Test**
```typescript
test('should update assigned_employees on date deletion', async () => {
  const date = await createTestDate();
```

### Test Utilities to Create

```typescript
// tests/helpers/constraint-test-helpers.ts

export async function expectConstraintViolation(
  operation: () => Promise<any>,
  constraintName: string
) {
  try {
    await operation();
    throw new Error('Expected constraint violation but operation succeeded');
  } catch (err) {
    expect(err.message).toContain(constraintName);
  }
}

export async function expectUniqueViolation(
  operation: () => Promise<any>
) {
  try {
    await operation();
    throw new Error('Expected unique violation but operation succeeded');
  } catch (err) {
    expect(err.code).toBe('23505');
  }
}

export async function testTransactionRollback(
  operation: () => Promise<any>,
  verifyRollback: () => Promise<void>
) {
  try {
    await operation();
    throw new Error('Expected transaction to fail');
  } catch (err) {
    // Verify rollback occurred
    await verifyRollback();
  }
}

export async function testConcurrentWrites(
  operation: () => Promise<any>,
  concurrency: number
) {
  const results = await Promise.allSettled(
    Array.from({ length: concurrency }, () => operation())
  );
  
  const succeeded = results.filter(r => r.status === 'fulfilled').length;
  const failed = results.filter(r => r.status === 'rejected').length;
  
  return { succeeded, failed };
}
```
