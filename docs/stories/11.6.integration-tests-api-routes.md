## Context

The application has multiple API routes that need integration testing:

**Employee Routes:**
- POST /api/employees (create with validations)
- GET /api/employees (list with filters)
- GET /api/employees/[id] (single employee)
- PATCH /api/employees/[id] (update with constraints)
- DELETE /api/employees/[id] (delete with cleanup)
- POST /api/employees/[id]/terminate (termination workflow)
- POST /api/employees/[id]/reactivate (reactivation workflow)
- POST /api/employees/export-crew-ready (export workflow)

**Important Dates Routes:**
- GET /api/important-dates (list all dates)
- POST /api/important-dates (create with capacity)
- PATCH /api/important-dates/[id] (update with validation)
- DELETE /api/important-dates/[id] (delete with cleanup)
- GET /api/important-dates/available-pe3 (PE3 uniqueness)

**Current State:**
- ❌ No systematic integration tests for API routes
- ✅ Some business logic tested in isolation (Stories 11.1-11.5)
- ❌ No tests for error handling (400, 404, 409, 500)
- ❌ No tests for authentication/authorization

### Key Test Patterns

**Pattern 1: CRUD Test**
```typescript
test('POST /api/employees creates employee with valid data', async () => {
  const employee = createTestEmployee();
  
  const response = await fetch('/api/employees', {
    method: 'POST',
    body: JSON.stringify(employee)
  });
  
  expect(response.status).toBe(201);
  const created = await response.json();
  expect(created.first_name).toBe(employee.first_name);
  expect(created.id).toBeDefined();
});
```

**Pattern 2: Validation Test**
```typescript
test('POST /api/employees returns 400 for invalid enum', async () => {
  const employee = createTestEmployee({ gender: 'male' }); // Invalid
  
  const response = await fetch('/api/employees', {
    method: 'POST',
    body: JSON.stringify(employee)
  });
  
  expect(response.status).toBe(400);
  const error = await response.json();
  expect(error.message).toContain('ogiltig kön');
  expect(error.field).toBe('gender');
});
```

**Pattern 3: Business Logic Integration Test**
```typescript
test('POST /api/employees decrements capacity on date assignment', async () => {
  const date = await createTestDate({ remaining_spots: 10 });
  const employee = createTestEmployee({ omc_date: date.date_value });
  
  const response = await fetch('/api/employees', {
    method: 'POST',
    body: JSON.stringify(employee)
  });
  
  expect(response.status).toBe(201);
  
  // Verify capacity decremented
  const updatedDate = await getDateById(date.id);
  expect(updatedDate.remaining_spots).toBe(9);
});
```

**Pattern 4: Error Handling Test**
```typescript
test('PATCH /api/employees/[id] returns 404 if not found', async () => {
  const response = await fetch('/api/employees/non-existent-id', {
    method: 'PATCH',
    body: JSON.stringify({ first_name: 'Updated' })
  });
  
  expect(response.status).toBe(404);
  const error = await response.json();
  expect(error.message).toContain('Anställd hittades inte');
});
```

**Pattern 5: Real-time Sync Test**
```typescript
test('employee creation triggers real-time update', async () => {
  const updates = [];
  const subscription = setupRealTimeSubscription('employees', (payload) => {
    updates.push(payload);
  });
  
  const employee = createTestEmployee();
  await fetch('/api/employees', {
    method: 'POST',
    body: JSON.stringify(employee)
  });
  
  // Wait for real-time update
  await waitFor(() => updates.length > 0, { timeout: 2000 });
  
  expect(updates[0].eventType).toBe('INSERT');
  expect(updates[0].new.first_name).toBe(employee.first_name);
  
  subscription.unsubscribe();
});
```

```typescript
// tests/helpers/api-test-helpers.ts

export async function makeAPIRequest(
  method: string,
  endpoint: string,
  body?: any
) {
  const response = await fetch(`http://localhost:3000${endpoint}`, {
    method,
    headers: {
      'Content-Type': 'application/json',
      // Add authentication headers if needed
    },
    body: body ? JSON.stringify(body) : undefined
  });
  
  return {
    status: response.status,
    body: await response.json()
  };
}

export function expectAPIError(
  response: Response,
  status: number,
  messageContains: string
) {
  expect(response.status).toBe(status);
  expect(response.body.message).toContain(messageContains);
}

export function setupRealTimeSubscription(
  table: string,
  callback: (payload: any) => void
) {
  const subscription = supabase
    .channel(`test-${table}`)
    .on('postgres_changes', { event: '*', schema: 'public', table }, callback)
    .subscribe();
  
  return subscription;
}

export async function cleanupTestData() {
  // Delete all test data created during tests
  await supabase.from('employees').delete().ilike('first_name', 'Test%');
  await supabase.from('important_dates').delete().ilike('date_description', 'Test%');
}
```
