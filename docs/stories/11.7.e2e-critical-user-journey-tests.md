# Story 11.7: End-to-End Critical User Journey Tests

## Metadata

- **Story ID**: 11.7
- **Epic**: Epic 11 - Comprehensive Test Coverage
- **Priority**: P1 (High)
- **Story Points**: 5
- **Status**: Proposed
- **Dependencies**: Stories 11.1-11.6 (E2E tests validate integrated system)

---

## User Story

**As a** QA engineer and product owner  
**I want** end-to-end tests for critical user journeys  
**So that** I can validate complete workflows from UI to database and catch integration issues before production

---

## Context

While unit and integration tests validate individual components, E2E tests validate complete user workflows through the entire stack (UI → API → Database → Real-time Updates).

**Critical User Journeys:**
1. Create employee → Assign dates → Export crew-ready
2. Terminate employee → Verify cleanup → Reactivate → Verify restoration
3. Import employees via CSV → Validate capacity respected → Export results
4. Complete prerequisites → Enable Crewing/Done → Export
5. Concurrent user scenarios (race conditions)

**E2E Framework:** Playwright (recommended) or Cypress

**Current State:** ❌ No E2E tests exist

---

### Key Test Patterns

**Pattern 1: Form Submission Test**
```typescript
test('employee lifecycle journey', async ({ page }) => {
  await page.goto('/employees');
  
  // Click add button
  await page.click('[data-testid="add-employee-btn"]');
  
  // Fill form
  await page.fill('[name="first_name"]', 'Anna');
  await page.fill('[name="surname"]', 'Test');
  await page.fill('[name="ssn"]', '199001011234');
  await page.selectOption('[name="rank"]', 'SEV');
  await page.selectOption('[name="gender"]', 'Kvinna');
  await page.fill('[name="hire_date"]', '2025-01-01');
  await page.fill('[name="omc_date"]', '8-9/3');
  
  // Submit
  await page.click('[type="submit"]');
  
  // Verify employee appears in table
  await expect(page.locator('table')).toContainText('Anna Test');
});
```

**Pattern 2: Modal Interaction Test**
```typescript
test('termination workflow', async ({ page }) => {
  // Navigate to employee detail
  await page.goto('/employees/emp-123');
  
  // Click terminate button
  await page.click('[data-testid="terminate-btn"]');
  
  // Verify modal appears
  await expect(page.locator('[role="dialog"]')).toBeVisible();
  
  // Review preview data
  await expect(page.locator('[data-testid="omc-repayment"]')).toContainText('8-9 mars 2025');
  
  // Confirm termination
  await page.click('[data-testid="confirm-terminate-btn"]');
  
  // Verify success message
  await expect(page.locator('[role="alert"]')).toContainText('Anställd avslutad');
});
```

**Pattern 3: Concurrent User Test**
```typescript
test('concurrent assignment to last spot', async ({ browser }) => {
  const contextA = await browser.newContext();
  const contextB = await browser.newContext();
  const pageA = await contextA.newPage();
  const pageB = await contextB.newPage();
  
  await pageA.goto('/employees/new');
  await pageB.goto('/employees/new');
  
  // Both users fill form
  await createEmployeeViaUI(pageA, { omc_date: '8-9/3' });
  await createEmployeeViaUI(pageB, { omc_date: '8-9/3' });
  
  // User A submits first
  await pageA.click('[type="submit"]');
  await expect(pageA.locator('[role="alert"]')).toContainText('Skapad');
  
  // User B submits (should fail)
  await pageB.click('[type="submit"]');
  await expect(pageB.locator('[role="alert"]')).toContainText('fullbokat');
  
  await contextA.close();
  await contextB.close();
});
```

**Pattern 4: Real-time Sync Test**
```typescript
test('real-time sync between users', async ({ browser }) => {
  const contextA = await browser.newContext();
  const contextB = await browser.newContext();
  const pageA = await contextA.newPage();
  const pageB = await contextB.newPage();
  
  // Both users view table
  await pageA.goto('/employees');
  await pageB.goto('/employees');
  
  const initialRowCount = await pageB.locator('table tbody tr').count();
  
  // User A creates employee
  await createEmployeeViaUI(pageA, { first_name: 'Anna' });
  
  // User B should see update within 2 seconds
  await expect(pageB.locator('table')).toContainText('Anna', { timeout: 2000 });
  
  const updatedRowCount = await pageB.locator('table tbody tr').count();
  expect(updatedRowCount).toBe(initialRowCount + 1);
  
  await contextA.close();
  await contextB.close();
});
```

```typescript
// tests/e2e/helpers/e2e-helpers.ts

export async function createEmployeeViaUI(page: Page, data: Partial<Employee>) {
  await page.fill('[name="first_name"]', data.first_name || 'Test');
  await page.fill('[name="surname"]', data.surname || 'Employee');
  await page.fill('[name="ssn"]', data.ssn || '199001011234');
  await page.selectOption('[name="rank"]', data.rank || 'SEV');
  await page.selectOption('[name="gender"]', data.gender || 'Man');
  await page.fill('[name="hire_date"]', data.hire_date || '2025-01-01');
  
  if (data.omc_date) {
    await page.fill('[name="omc_date"]', data.omc_date);
  }
}

export async function waitForTableUpdate(page: Page, expectedRowCount?: number) {
  if (expectedRowCount) {
    await page.waitForFunction(
      (count) => document.querySelectorAll('table tbody tr').length === count,
      expectedRowCount,
      { timeout: 3000 }
    );
  } else {
    await page.waitForTimeout(1000); // Generic wait for update
  }
}

export async function verifyCapacityBadge(
  page: Page,
  status: 'full' | 'almost-full' | 'available'
) {
  const badge = page.locator('[data-testid="capacity-badge"]');
  
  if (status === 'full') {
    await expect(badge).toHaveClass(/bg-red/);
    await expect(badge).toContainText('Fullbokad');
  } else if (status === 'almost-full') {
    await expect(badge).toHaveClass(/bg-yellow/);
    await expect(badge).toContainText('Nästan fullbokad');
  } else {
    await expect(badge).not.toBeVisible();
  }
}

export async function downloadAndParseCSV(page: Page): Promise<string[][]> {
  const downloadPromise = page.waitForEvent('download');
  await page.click('[data-testid="export-btn"]');
  const download = await downloadPromise;
  
  const path = await download.path();
  const csv = await fs.readFile(path, 'utf-8');
  
  return csv.split('\n').map(row => row.split(','));
}
```
