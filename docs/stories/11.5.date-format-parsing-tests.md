### ÖMC Date Format Rules (Story 8.9)

**Display Format:**
- Pattern: `{day1}-{day2} {month_swedish} {year}`
- Example: "8-9 mars 2025"
- Leading zeros: No (8, not 08)

**Input Formats Accepted:**
1. Short: "8-9/3" (assumes current year)
2. Medium: "8-9 mars" (assumes current year)
3. Full: "8-9 mars 2025"
4. ISO: "2025-03-08" (start date)

**Validation Rules:**
- Must be exactly 2 consecutive days
- Must be in the same month
- Must be in the same year
- Start date must come before end date

**Swedish Month Names:**
```typescript
const swedishMonths = [
  'januari', 'februari', 'mars', 'april',
  'maj', 'juni', 'juli', 'augusti',
  'september', 'oktober', 'november', 'december'
];
```

```typescript
// tests/helpers/omc-date-test-helpers.ts

export function createOMCDate(day: number, month: number, year: number) {
  return new Date(year, month - 1, day); // month is 0-indexed
}

export function expectConsecutiveDays(date1: Date, date2: Date) {
  const diff = Math.abs(date2.getTime() - date1.getTime());
  const daysDiff = diff / (1000 * 60 * 60 * 24);
  expect(daysDiff).toBe(1);
}

export function parseOMCDateVariants(date: Date) {
  const day1 = date.getDate();
  const day2 = day1 + 1;
  const month = date.getMonth() + 1;
  const year = date.getFullYear();
  const monthName = swedishMonths[date.getMonth()];
  
  return [
    `${day1}-${day2}/${month}`, // Short format
    `${day1}-${day2} ${monthName}`, // Medium format
    `${day1}-${day2} ${monthName} ${year}`, // Full format
    date.toISOString().split('T')[0] // ISO format
  ];
}

export function testAllOMCDateFormats(inputFormats: string[], expectedDate: string) {
  inputFormats.forEach(format => {
    const parsed = parseOMCDate(format, new Date().getFullYear());
    expect(parsed.toISOString().split('T')[0]).toBe(expectedDate);
  });
}
```
